# Development and test dependencies.
#
# Deliberately separate from `backend/requirements.txt` and
# `ml/requirements.txt`: nothing here ships in a runtime image. Keeping them
# apart is what lets the backend image stay small and free of test doubles - a
# serving container with `fakeredis` installed is one import away from silently
# running without a real cache.
#
#     pip install -r backend/requirements.txt -r ml/requirements.txt -r requirements-dev.txt

# --- Test runner --------------------------------------------------------------
pytest>=8.0,<10
pytest-cov>=5.0

# `TestClient` is a thin wrapper over httpx, so the API tests exercise the real
# ASGI stack - routing, dependencies, middleware - rather than calling handler
# functions directly.
httpx>=0.27,<1

# A Redis API-compatible double. The cache *logic* is therefore tested exactly
# as written, including hit ratios; only Redis network latency goes unmeasured,
# and `scripts/benchmark_serving.py` says so in its own output.
fakeredis>=2.23,<3

# --- Linting and typing -------------------------------------------------------
ruff>=0.5
mypy>=1.10

# deployment target uses Compose. If you need a local database and cannot run
